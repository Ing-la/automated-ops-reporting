```

│
├── src/                      # 源代码
│   ├── processing/           # 数据处理
│   ├── metrics/              # 指标计算
│   ├── visualization/        # 图表生成
│   ├── llm/                  # LLM分析
│   ├── report/               # 报告生成
│   ├── delivery/             # 推送上传
│   └── utils/                # 工具函数
│
├── scripts/                   # 执行脚本

└── output/                    # 输出目录
```
